async function addProduct(){ const title=pt.value.trim(); const upc=pu.value.trim(); const price=Number(pp.value); if(!title||!upc||!price){ alert('Enter the product title, UPC, and Walmart price.'); return; } const productData={ title, upc, walmart_price:price, oos_status:ps.value, bestseller_rank:Number(pr.value)||null, review_count:Number(pv.value)||null }; try{ const response=await fetch('/api/products',{ method:'POST', headers:{ 'content-type':'application/json' }, body:JSON.stringify(productData) }); const result=await response.json(); if(!response.ok||!result.ok){ throw new Error( result.error|| 'The product could not be saved.' ); } products.push({ id:result.product_id, title, upc, price, status:ps.value, rank:Number(pr.value)||0, reviews:Number(pv.value)||0 }); save(); pt.value=''; pu.value=''; pp.value=''; pr.value=''; pv.value=''; modal.style.display='none'; await scan(); alert( 'Product saved permanently to SourceMatch.' ); }catch(error){ alert( 'SourceMatch could not save the product: '+ error.message ); } }
